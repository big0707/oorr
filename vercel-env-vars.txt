# Vercel 环境变量配置
# 复制以下内容到 Vercel Dashboard > Settings > Environment Variables
# 
# 使用方法：
# 1. 访问 https://vercel.com/dashboard
# 2. 选择你的项目
# 3. 进入 Settings > Environment Variables
# 4. 逐个添加以下变量（Production, Preview, Development 都要添加）

# ============================================
# 前端环境变量（必须以 VITE_ 开头）
# ============================================

# ⚠️ 重要：请替换为你的真实值
# 从 Firebase Console > Project Settings > General > Your apps 中获取
VITE_FIREBASE_API_KEY=your-firebase-api-key-here
VITE_FIREBASE_AUTH_DOMAIN=your-project.firebaseapp.com
VITE_FIREBASE_PROJECT_ID=your-project-id
VITE_FIREBASE_STORAGE_BUCKET=your-project.appspot.com
VITE_FIREBASE_MESSAGING_SENDER_ID=your-sender-id
VITE_FIREBASE_APP_ID=your-app-id

# ============================================
# 后端环境变量（用于 Serverless Functions）
# ============================================

# FIREBASE_CREDENTIALS_JSON 需要从 firebase-service-account.json 文件获取
# 运行以下命令生成压缩后的 JSON：
# cat firebase-service-account.json | jq -c .
# 或者手动复制 firebase-service-account.json 的内容并压缩为一行
FIREBASE_CREDENTIALS_JSON={"type":"service_account","project_id":"your-project-id","private_key_id":"...","private_key":"...","client_email":"...","client_id":"...","auth_uri":"...","token_uri":"...","auth_provider_x509_cert_url":"...","client_x509_cert_url":"...","universe_domain":"googleapis.com"}
