# Vercel 环境变量配置
# 在 Vercel Dashboard > Settings > Environment Variables 中逐个添加
# 每个变量都要添加到 Production, Preview, Development 三个环境
#
# ⚠️ 请替换为你的真实值（从 Firebase Console 和 Supabase Dashboard 获取）

# --- Firebase 前端（必须以 VITE_ 开头）---
VITE_FIREBASE_API_KEY=your-firebase-api-key
VITE_FIREBASE_AUTH_DOMAIN=your-project.firebaseapp.com
VITE_FIREBASE_PROJECT_ID=your-project-id
VITE_FIREBASE_STORAGE_BUCKET=your-project.firebasestorage.app
VITE_FIREBASE_MESSAGING_SENDER_ID=your-sender-id
VITE_FIREBASE_APP_ID=your-app-id

# --- Firebase 后端 ---
# 运行: cat firebase-service-account.json | jq -c .
# 将输出的一整行 JSON 粘贴到这里
FIREBASE_CREDENTIALS_JSON=<paste-compressed-json-here>

# --- Supabase ---
SUPABASE_URL=https://your-project.supabase.co
SUPABASE_ANON_KEY=your-anon-key
VITE_SUPABASE_URL=https://your-project.supabase.co
VITE_SUPABASE_ANON_KEY=your-anon-key
